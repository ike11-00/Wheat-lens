# Leaf Lens - Python dependencies
#
#   python -m venv .venv
#   source .venv/bin/activate          # Windows: .venv\Scripts\activate
#   pip install -r requirements.txt
#
# ---------------------------------------------------------------------------
# Supported Python versions
# ---------------------------------------------------------------------------
# TensorFlow is the only dependency that constrains the Python version; every
# other package here already ships wheels for 3.13+.
#
#   Python 3.10 - 3.12   supported (TensorFlow 2.17 - 2.21)
#   Python 3.13          supported (needs TensorFlow >= 2.20)
#   Python 3.14          NOT supported - no TensorFlow release ships a cp314
#                        wheel yet. Use 3.13 or earlier until one does.
#
# Verified in this project: the full test suite (90 tests), model loading and
# the web application all pass on TensorFlow 2.17.1 (NumPy 1.26) and on
# TensorFlow 2.21.0 (NumPy 2.4). A model trained under 2.17 loads under 2.21
# and produces bit-identical predictions.
#
# If you need a specific Python version, python.org keeps every release
# archived, and pyenv / conda / uv can install any of them:
#   pyenv install 3.12.8   |   conda create -n leaflens python=3.12
# ---------------------------------------------------------------------------

# --- machine learning -------------------------------------------------
# Lower bound: the oldest version this project was tested against.
# Upper bound: excludes a hypothetical 3.x with breaking API changes.
# GPU users: replace tensorflow-cpu with `tensorflow`.
tensorflow-cpu>=2.17,<3.0
scikit-learn>=1.4,<2.0        # metrics, confusion matrix, classification report

# NumPy is deliberately NOT pinned here. TensorFlow declares its own
# compatible range (2.17 requires <2.0, 2.20+ accepts 2.x), so pinning it
# ourselves would silently force an old TensorFlow and block Python 3.13.
numpy>=1.26

# --- images -----------------------------------------------------------
Pillow>=10.0                  # image decoding, validation, resizing
ImageHash>=4.3                # perceptual hashing for near-duplicate detection

# --- data and plotting ------------------------------------------------
pandas>=2.0                   # tabular handling in reports
matplotlib>=3.8               # training curves, distributions, grids
seaborn>=0.13                 # confusion-matrix heat maps
PyYAML>=6.0                   # configuration files

# --- web application --------------------------------------------------
Flask>=3.0                    # prediction API and interface

# --- development ------------------------------------------------------
pytest>=8.0                   # test suite

# --- optional ---------------------------------------------------------
